% ---------------------------------------------------------------- why
\begin{frame}
  \frametitle{Why Prove It? Databases Rewrite Your Questions}
  \begin{itemize}
    \item To answer fast, a database rewrites your question into a quicker
      one, which must have the same answer.
    \item Each rewrite follows a \textbf{law}, such as ``filter first, then
      combine''. Some laws hold only under a side condition.
    \item Get one wrong, and the database silently answers wrongly. Testing
      tools have found hundreds of such bugs in mature databases.
  \end{itemize}
  \begin{sticky}[halign = center]
    A law can be \textbf{proved}. In this challenge, you prove them.
  \end{sticky}
  \slideref{SQLancer: \url{https://github.com/sqlancer/sqlancer} \rfsep
    M. Rigger, Z. Su. \textbf{Detecting Optimization Bugs in Database Engines via Non-Optimizing Reference Engine Construction}. ESEC/FSE 2020.}
\end{frame}
