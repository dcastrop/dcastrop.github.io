% Level 5: fast joins, and the Lean they need
\begin{frame}
  \frametitle{Fast Joins Must Compute the Same Relation}
  \begin{itemize}
    \item \textbf{The definition}: for each row of $R$, for each row of
      $S$, keep the pairs that agree: about $|R| \times |S|$ steps.
    \item \textbf{Hash join}: put the rows of $S$ in a hash map, keyed by
      the shared attributes; look up the partners of each row of $R$:
      about $|R| + |S|$ steps.
    \item Prove that each computes exactly $R \join S$.
  \end{itemize}
  \begin{sticky}
    ``Future users of large data banks must be protected from having to
    know how the data is organized in the machine.''\hfill --- E. F. Codd,
    1970
  \end{sticky}
  \slideref{E. F. Codd. \textbf{A Relational Model of Data for Large Shared Data Banks}. CACM 13(6), 1970. \rfsep
    \lnk{ahv6} (section 6.1)}
  \leveltag{5}
\end{frame}
