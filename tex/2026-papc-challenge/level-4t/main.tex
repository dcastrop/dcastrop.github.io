% ---------------------------------------------------------------- theory
\theorypart{Products, images, and a side condition}

\begin{frame}
  \frametitle{A Product Pairs Every Row with Every Row}
  \defn{Definition 4T.1 (Product)}{For tables $R$ and $S$:
    \ $R \times S = \{(t, s) \mid t \in R \text{ and } s \in S\}$.}
  Its rows are pairs of rows. For example,
  $((\text{Ada}, \text{Maths}), (1, \text{Lists})) \in \rel{Member} \times
  \rel{Session}$, which has $3 \cdot 4 = 12$ rows. In SQL:
  \code{FROM Member, Session}.
  \defn{Definition 4T.2 (Image, projection)}{For a function $f$ on rows,
    $f[R] = \{s \mid \text{there is } t \in R \text{ with } f(t) =
    s\}$. The projections are the images under $\pi_1(d_1, d_2) = d_1$
    and $\pi_2(d_1, d_2) = d_2$.}
  For example, $\pi_2[\rel{Attended}] = \{1, 2, 3\}$: the weeks someone
  attended.
  \slideref{\lnk{ahv4}}
  \theorytag{4T}
\end{frame}

\begin{frame}
  \frametitle{Images Compose}
  \defn{Fact 4T.3 (Composition)}{$g[f[R]] = (g \circ f)[R]$.}
  \emph{Proof.} For every $s$:
  \[
    \begin{array}{@{}r@{\quad}l@{}}
      s \in g[f[R]] & \iff \text{there is } u \in f[R] \text{ with }
        s = g(u) \\
      & \iff \text{there is } t \in R \text{ with } s = g(f(t)) \\
      & \iff s \in (g \circ f)[R]. \\
    \end{array}
  \]
  \proofend\par
  In the middle step: from left to right, take apart the two ``there
  is''; from right to left, the witness for $u$ is $f(t)$.
  \theorytag{4T}
\end{frame}

\begin{frame}
  \frametitle{A Side Condition Is a Hypothesis of the Law}
  \defn{Definition 4T.4 (Mentions only $R$)}{A condition $p$ on the rows
    of $R \times S$ \emph{mentions only $R$} if there is a condition $p'$
    on the rows of $R$ with: for all $t$ and $s$, $p(t, s)$ holds
    exactly when $p'(t)$ holds.}
  \defn{Law 5}{If $p$ mentions only $R$, then $\sel{p}(R \times S) =
    \sel{p'}(R) \times S$.}
  \emph{Proof.} $(t, s) \in \sel{p}(R \times S) \iff t \in R$, $s \in S$
  and $p(t, s)$ $\iff t \in R$, $p'(t)$ and $s \in S$ $\iff (t, s) \in
  \sel{p'}(R) \times S$. \proofend
  \par\medskip
  Without the hypothesis, the law makes no sense: a condition on the week
  of a session cannot be checked on $\rel{Member}$ alone.
  \slideref{\lnk{ahv6} (section 6.1)}
  \theorytag{4T}
\end{frame}
