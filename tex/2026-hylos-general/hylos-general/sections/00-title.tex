\begin{frame}
  \frametitle{This Talk}
  \begin{center}
    \begin{tikzpicture}[alt={A clear but slow specification is turned into
        a fast program by rewriting with proved laws.},

        minimum height = 12mm}]
      \node[box, stage] (s) {\textbf{clear, slow}\\ specification};
      \node[hot, stage, right = 46mm of s] (p) {\textbf{fast}\\ program};
      \draw[->, very thick] (s) -- node[above, note, align = center]
        {rewrite with\\ \emph{laws}} (p);
    \end{tikzpicture}
  \end{center}
  \begin{enumerate}
    \item \textbf{Program calculation}: how do laws turn a clear
      specification into a fast program?\\
      {\small\textcolor{black!60}{The classic theory, and our ICFP 2016
      work on parallel programs.}}
    \item \textbf{Mechanising it}: can a machine check every step, side
      conditions included, and still give clean code?\\
      {\small\textcolor{black!60}{Our Rocq library: joint work with
      M. Paviotti and M. Vollmer (ITP 2025).}}
    \item \textbf{Low-level calculation} (ongoing): how far down can we
      go? Can untrusted search (compilers, e-graphs, ML) propose the steps,
      with Rocq checking each one?
  \end{enumerate}
  \speakernote{Answer to the last slide: laws, proved, side conditions
    checked \\
    Theme: spec in, fast program out, every step an equation \\
    1: the classic theory, with code; laws for parallelism (ours, ICFP
    2016) \\
    2: making it machine-checked, with runnable code (ours, with Paviotti
    and Vollmer) \\
    3: where we are going; search proposes, Rocq checks}
\end{frame}
